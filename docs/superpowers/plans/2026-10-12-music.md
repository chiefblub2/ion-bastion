# Music from Game Sounds Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make the game's sounds play as music: cues quantized to a per-theme beat grid and pitched to a chord loop; enemies drive the rhythm section (threat), towers drive harmony and an arpeggiator (defense), and the clear time between them colours the mood.

**Architecture:** Pure modules `clock` (beat grid), `harmony` (theme music, chords, voicing), `intensity` (threat, defense, clear time, colour), `grooves` (drum parts, arpeggio) and `sequencer` (lookahead window, "A Tale of Two Clocks"), plus `score` (places a cue on the grid at a chord tone). `cues.ts` hands out chord-tone voices; `engine` schedules notes ahead; `ambience` follows chords and tone. `app/audio.ts` owns score, sequencer and the intensity refresh. Simulation untouched.

**Tech Stack:** TypeScript strict ESM, Web Audio API, Vitest 3 (Node).

**Spec:** `docs/superpowers/specs/2026-10-12-music-design.md` (including "Two parts: enemies and towers"; builds on `docs/superpowers/specs/2026-10-12-sound-design-design.md`)

**Spec refinements in this plan (same data):** grid and pitch per cue live in a `TIMING` table next to `RECIPES` instead of inside `CueSpec`; `cuesFor` hands out a chord-tone `voice` and `Score` turns it into a playback rate at the scheduled time.

## Global Constraints

- Render side only: no change to `src/core`, `src/systems`, `src/content`, state, hash or network. `intensity.ts` only reads state and calls the pure `effectiveTowerStats` (without `sim`).
- No `AudioContext` and no DOM at import time; pure modules run in Vitest/Node.
- Lookahead scheduling: 25 ms timer, 100 ms window on the audio clock.
- Lag ≤ one grid step; `reject`, `click`, `hover` instant.
- Pitched playback rate within 0.7–1.42 of the recipe (nearest-octave snapping).
- Thresholds: threat full at 30, defense full at 40; drums hat / kick ≥ 0.25 / snare ≥ 0.5 / 16th hats + bass ≥ 0.75; harmony 7ths ≥ 0.25, 9ths ≥ 0.5, arpeggio ≥ 0.75; colour bright < 3 s (leave 4), dark > 10 s (leave 8); heavy ≥ 300 base HP, boss ≥ 1000.
- Drums pool cap 10, on the music bus.
- Files < 500 lines; English; `npm test` and `npm run build` pass at the end. Between Task 6 and Task 9 `tsc` fails (the `Cue` shape changes before its callers); those tasks run Vitest only; Task 9 restores `tsc`.
- Notes and review results to `/Volumes/T7 Shield/Temp/ion-bastion-sound/`.

## Review Focus

1. **Mission change mid-bar:** the new tempo starts on the next bar line, not mid-bar → `Score` test (Task 6).
2. **Muted or hidden (audio clock frozen):** the sequencer never schedules a moment twice → `Sequencer` test (Task 4).
3. **Long frame:** a cue whose slot passed plays at the next slot after now → `clock` test (Task 1).
4. **Every pitched cue in every theme, bar and harmony extent** stays within 0.7–1.42 → sweep test (Task 6).
5. **Colour flicker** when the clear time hovers at a threshold → hysteresis test (Task 3).

```ts
// The music's beat grid on the audio clock (seconds of AudioContext.currentTime). 4/4 time.

/** Note value of a grid step: 32 = 1/32, 16 = 1/16, 8 = 1/8, 4 = a beat, 1 = a bar; 0 plays at once. */
export type Grid = 32 | 16 | 8 | 4 | 1 | 0;
export interface Clock {
  /** Audio time of bar 0. */
  origin: number;
  bpm: number;
}
/** Small tolerance, so a time exactly on a slot stays on it despite float error. */
const EPS = 1e-9;
export const stepLength = (c: Clock, grid: Exclude<Grid, 0>) => ((60 / c.bpm) * 4) / grid;

/** The first slot of `grid` at or after `now`; instant cues play at `now`. */
export function nextSlot(c: Clock, now: number, grid: Grid) {
  if (grid === 0) return now;
  const len = stepLength(c, grid);
  return c.origin + Math.ceil((now - c.origin) / len - EPS) * len;
}
export const barAt = (c: Clock, time: number) => Math.floor((time - c.origin) / stepLength(c, 1) + EPS);
export const stepAt = (c: Clock, time: number) => Math.floor((time - c.origin) / stepLength(c, 16) + EPS);
/** A new tempo that starts at the next bar line of the old one, so the music never stumbles mid-bar. */
export const rebase = (c: Clock, now: number, bpm: number): Clock => ({ origin: nextSlot(c, now, 1), bpm });
```

```ts
import { fnv } from "../core/hash";
import type { DroneId } from "./mood";
// Key, tempo and a 4-bar chord loop per theme. The defense widens the chords; a dark mood adds a tension note.

export const MODES = {
  ionian: [0, 2, 4, 5, 7, 9, 11],
  dorian: [0, 2, 3, 5, 7, 9, 10],
  phrygian: [0, 1, 3, 5, 7, 8, 10],
  lydian: [0, 2, 4, 6, 7, 9, 11],
  mixolydian: [0, 2, 4, 5, 7, 9, 10],
  aeolian: [0, 2, 3, 5, 7, 8, 10],
  harmonic: [0, 2, 3, 5, 7, 8, 11],
} as const;
export type Mode = keyof typeof MODES;
export type Groove = "none" | "straight" | "half" | "broken";
/** The mood from the balance between threat and defense. */
export type Color = "bright" | "drive" | "dark";
/** Chord size: triad, 7th or 9th. */
export type Extent = 3 | 4 | 5;
export interface ThemeMusic {
  bpm: number;
  /** MIDI note of the key's root (the drone's root). */
  root: number;
  mode: Mode;
  /** Scale degree (0 = tonic) of the chord in each of the four bars. */
  progression: readonly [number, number, number, number];
  groove: Groove;
}
const m = (bpm: number, root: number, mode: Mode, progression: [number, number, number, number], groove: Groove): ThemeMusic => ({ bpm, root, mode, progression, groove });

export const MUSIC: Record<DroneId, ThemeMusic> = {
  menu: m(84, 33, "aeolian", [0, 5, 2, 6], "none"),
  outpost: m(112, 33, "aeolian", [0, 5, 2, 6], "straight"),
  lock: m(108, 34, "dorian", [0, 3, 0, 6], "straight"),
  shard: m(118, 35, "lydian", [0, 1, 4, 0], "broken"),
  ember: m(124, 31, "phrygian", [0, 1, 0, 6], "broken"),
  core: m(120, 33, "harmonic", [0, 5, 3, 4], "straight"),
  frost: m(100, 36, "dorian", [0, 3, 6, 4], "half"),
  toxic: m(116, 32, "phrygian", [0, 1, 6, 0], "broken"),
  orbit: m(96, 38, "lydian", [0, 1, 0, 4], "half"),
  ruin: m(104, 30, "aeolian", [0, 3, 5, 4], "half"),
  rift: m(126, 28, "phrygian", [0, 1, 3, 1], "broken"),
  dune: m(110, 33, "dorian", [0, 6, 3, 0], "straight"),
  abyss: m(92, 26, "aeolian", [0, 5, 6, 0], "half"),
  storm: m(128, 31, "aeolian", [0, 6, 5, 4], "broken"),
  jungle: m(114, 35, "mixolydian", [0, 6, 3, 0], "straight"),
  volcano: m(122, 29, "harmonic", [0, 5, 1, 4], "straight"),
  geode: m(106, 37, "lydian", [0, 4, 1, 0], "half"),
  gear: m(120, 34, "mixolydian", [0, 6, 3, 4], "straight"),
  tide: m(102, 32, "ionian", [0, 5, 3, 4], "half"),
};

/** MIDI notes of the bar's chord, stacked in thirds from its degree: triad, + 7th, + 9th. */
export function chordOf(music: ThemeMusic, bar: number, extent: Extent = 3): number[] {
  const scale = MODES[music.mode],
    degree = music.progression[((bar % 4) + 4) % 4];
  return [0, 2, 4, 6, 8].slice(0, extent).map((s) => {
    const i = degree + s;
    return music.root + scale[i % 7] + 12 * Math.floor(i / 7);
  });
}
/** The defense builds the harmony: triads, then 7ths from a quarter, 9ths from half. */
export const extentFor = (defense: number): Extent => (defense >= 0.5 ? 5 : defense >= 0.25 ? 4 : 3);
/** The pad's chord: a dark mood adds the minor second above the root as a tension note. */
export const padChord = (chord: readonly number[], color: Color) => (color === "dark" ? [...chord, chord[0] + 13] : [...chord]);
/** MIDI note per drone voice: the bass on the root, the other voices from the top of the chord down, an octave up. */
export function padVoicing(chord: readonly number[], voices: number) {
  const n = chord.length;
  return Array.from({ length: voices }, (_, k) => (k === 0 ? chord[0] : chord[(((n - k) % n) + n) % n] + 12));
}
export const hz = (midi: number) => 440 * 2 ** ((midi - 69) / 12);
/** Chord tone `index` (wrapping) in the octave whose pitch lies nearest `targetHz`. */
export function snap(targetHz: number, chord: readonly number[], index: number) {
  const note = chord[((index % chord.length) + chord.length) % chord.length],
    target = 69 + 12 * Math.log2(targetHz / 440);
  return hz(note + 12 * Math.round((target - note) / 12));
}
/** A stable chord-tone index per tower, so each tower keeps its own voice. */
export const voiceOf = (tower: string) => fnv(`voice:${tower}`) % 4;
```

```ts
import type { ContentPack, GameEvent, GameState } from "../core/types";
import { isSupport } from "../systems/attacks";
import { effectiveTowerStats } from "../systems/auras";
import type { Color } from "./harmony";
// The two parts of the music: the enemies' threat (rhythm), the defense's strength (harmony), and the clear time
// between them (mood). Reads state only.

/** Base HP from which an enemy stomps on the beat, and from which it is a boss with its own motif. */
export const HEAVY_HP = 300,
  BOSS_HP = 1000;
/** Weighted enemies and tower levels at which threat and defense are full. */
const FULL_THREAT = 30,
  FULL_DEFENSE = 40;
export interface Intensity {
  threat: number;
  defense: number;
  /** Seconds the towers need to clear the field at their combined damage per second. */
  clear: number;
  heavies: number;
}
export function intensityOf(state: GameState, content: ContentPack, pathLength: number): Intensity {
  let threat = 0,
    hp = 0,
    heavies = 0,
    defense = 0,
    dps = 0;
  for (const e of state.enemies) {
    const base = content.enemies[e.type].hp,
      progress = pathLength > 0 ? Math.min(1, e.distance / pathLength) : 0;
    threat += Math.min(4, Math.max(0.5, base / 100)) * (0.5 + 0.5 * progress);
    hp += e.hp;
    if (base >= HEAVY_HP) heavies++;
  }
  for (const t of state.towers) {
    defense += 1 + t.upgrades.length;
    if (isSupport(content.towers[t.type].attack)) continue;
    const s = effectiveTowerStats(t, state.towers, content);
    if (s.interval > 0) dps += s.damage / s.interval;
  }
  return { threat: Math.min(1, threat / FULL_THREAT), defense: Math.min(1, defense / FULL_DEFENSE), clear: hp / Math.max(1, dps), heavies };
}
/** Bright under 3 s, dark over 10 s; each colour holds a little past its threshold so it does not flicker. */
export function colorOf(clear: number, previous: Color): Color {
  if (clear < (previous === "bright" ? 4 : 3)) return "bright";
  if (clear > (previous === "dark" ? 8 : 10)) return "dark";
  return "drive";
}
/** Pad filter factor: brighter with a comfortable lead and with a stronger defense. */
export const toneFor = (color: Color, defense: number) => (color === "bright" ? 1.5 : color === "dark" ? 0.6 : 1) * (1 + 0.5 * defense);
export function bossSpawned(events: readonly GameEvent[], state: GameState, content: ContentPack) {
  return events.some((e) => {
    if (e.type !== "spawn") return false;
    const enemy = state.enemies.find((x) => x.id === e.enemy);
    return !!enemy && content.enemies[enemy.type].hp >= BOSS_HP;
  });
}
```

```ts
import { nextSlot, stepLength, type Clock } from "./clock";
import { arpIndex, hitsAt, type Drum } from "./grooves";
import { chordOf, padChord, type Color, type Extent, type ThemeMusic } from "./harmony";
// Lookahead scheduling after Chris Wilson's "A Tale of Two Clocks": a coarse timer schedules everything due in the
// next window on the precise audio clock.

const TICK_MS = 25,
  AHEAD = 0.1;
/** What the two parts of the music play right now. */
export interface Parts {
  drums: readonly Drum[];
  arp: boolean;
  extent: Extent;
  color: Color;
}
export type Beat = { kind: "chord"; notes: number[]; at: number } | { kind: "hit"; drum: Drum; at: number } | { kind: "arp"; voice: number; at: number };

/** The 1/16 steps with `from <= time < to`. */
export function stepsIn(c: Clock, from: number, to: number) {
  const len = stepLength(c, 16),
    out: { time: number; step: number }[] = [];
  for (let step = Math.round((nextSlot(c, from, 16) - c.origin) / len); ; step++) {
    const time = c.origin + step * len;
    if (time >= to - 1e-9) return out;
    out.push({ time, step });
  }
}
/** Pad chords on bar lines, drum hits and arpeggio notes in the window. */
export function schedule(c: Clock, music: ThemeMusic, parts: Parts, from: number, to: number): Beat[] {
  const out: Beat[] = [];
  for (const { time, step } of stepsIn(c, from, to)) {
    if (((step % 16) + 16) % 16 === 0)
      out.push({ kind: "chord", notes: padChord(chordOf(music, Math.floor(step / 16), parts.extent), parts.color), at: time });
    for (const drum of hitsAt(music.groove, step, parts.drums)) out.push({ kind: "hit", drum, at: time });
    if (parts.arp) out.push({ kind: "arp", voice: arpIndex(step, parts.color), at: time });
  }
  return out;
}
export class Sequencer {
  private until = -Infinity;
  private timer: ReturnType<typeof setInterval> | null = null;
  constructor(
    private now: () => number,
    private window: (from: number, to: number) => void,
  ) {}
  start() {
    this.timer ??= setInterval(() => this.tick(), TICK_MS);
  }
  stop() {
    if (this.timer) clearInterval(this.timer);
    this.timer = null;
  }
  /** Hands out the part of the next window not scheduled yet; a frozen clock (suspended context) yields nothing. */
  tick() {
    const now = this.now(),
      from = Math.max(this.until, now),
      to = now + AHEAD;
    if (to <= from) return;
    this.window(from, to);
    this.until = to;
  }
}
```

```ts
// accompany it and stay silent. `Score.place` decides when and at which pitch a cue sounds.

export interface Cue {
  id: CueId;
  /** Chord-tone index for pitched cues: the tower's own voice, or a fixed tone for flow cues. */
  voice: number;
  gain: number;
}
export type View = "menu" | "game";
/** Support kinds never fire, so they have no cue. */
const FIRE: Record<AttackSpec["kind"], CueId | null> = {
  direct: "bolt", pierce: "beam", volley: "bolt", splash: "lob", mortar: "lob", chain: "crackle", focus: "beam", quake: "swell",
  slow: "chime", net: "chime", burn: "hiss", corrode: "hiss", decay: "hiss", stun: "zap", disrupt: "zap", pull: "sweep",
  execute: "crack", bleed: "trap", charge: "trap", pit: "trap", alarm: "trap",
  aura: null, income: null, detect: null, bounty: null, repair: null, mark: null,
};
const BLASTS = new Set<AttackSpec["kind"]>(["splash", "mortar", "charge"]);
const PLAIN: Partial<Record<GameEvent["type"], CueId>> = {
  kill: "kill", evade: "evade", overload: "overload", leak: "leak", build: "build", sell: "sell", upgrade: "upgrade",
  income: "income", repair: "repair", waveStart: "waveStart", waveEnd: "waveEnd",
};
/** Building rises to the fifth, upgrading to the top of the chord; other flow cues sit on the root. */
const FLOW_VOICE: Partial<Record<CueId, number>> = { build: 2, upgrade: 3 };
export const burstGain = (n: number) => Math.min(1.5, 1 + 0.1 * (n - 1));

function cueOf(e: GameEvent, content: ContentPack): { id: CueId; voice: number } | null {
  switch (e.type) {
    case "shot": {
      const d = content.towers[e.tower];
      const id = d.placement === "path" ? "trap" : FIRE[d.attack.kind];
      return id && { id, voice: voiceOf(e.tower) };
    }
    case "impact":
      return { id: BLASTS.has(content.towers[e.tower].attack.kind) ? "blast" : "impact", voice: voiceOf(e.tower) };
    case "end":
      return { id: e.result, voice: 0 };
    default: {
      const id = PLAIN[e.type];
      return id ? { id, voice: FLOW_VOICE[id] ?? 0 } : null;
    }
  }
}
/** One cue per sound and voice in this frame; in the menu only flow cues (a multiplayer wave runs on). */
export function cuesFor(events: readonly GameEvent[], content: ContentPack, view: View): Cue[] {
  const merged = new Map<string, { id: CueId; voice: number; n: number }>();
  for (const e of events) {
    const c = cueOf(e, content);
    if (!c || (view === "menu" && RECIPES[c.id].pool === "combat")) continue;
    const key = `${c.id}@${c.voice}`,
      m = merged.get(key);
    if (m) m.n++;
    else merged.set(key, { ...c, n: 1 });
  }
  return [...merged.values()].map(({ id, voice, n }) => ({ id, voice, gain: burstGain(n) }));
}
/** Reports the reactor falling to 30 % once per crossing; a new state (restart, mission change) starts fresh. */
export class ReactorWatch {
  private seen: { state: GameState; lives: number } | null = null;
  crossed(state: GameState, max: number) {
    const before = this.seen?.state === state ? this.seen.lives : undefined;
    this.seen = { state, lives: state.lives };
    return before !== undefined && max > 0 && state.lives > 0 && before / max > 0.3 && state.lives / max <= 0.3;
  }
}
/** Seats whose elimination tick appeared since the last frame, without the local player (the `lost` cue covers them). */
export function newlyEliminated(before: readonly (number | null)[], after: readonly (number | null)[], local: number) {
  return after.flatMap((tick, i) => (tick !== null && (before[i] ?? null) === null && i !== local ? [i] : []));
}
```

```ts
  private kills = 0;
  /** Switches the theme's music; a new tempo starts on the next bar line. True when the theme changed. */
  theme(drone: DroneId, now: number) {
    if (drone === this.drone) return false;
    this.music = MUSIC[drone];
    this.clock = this.drone === null ? { origin: now, bpm: this.music.bpm } : rebase(this.clock, now, this.music.bpm);
    this.drone = drone;
    return true;
  }
  /** The harmonic chord at `time` (without the pad's tension note, so shots stay in tune). */
  chordAt(time: number) {
    return chordOf(this.music, barAt(this.clock, time), this.extent);
  }
  noteAt(id: CueId, voice: number, gain: number, at: number): Note {
    if (!TIMING[id].pitched) return { id, rate: 1, gain, at };
    const base = RECIPES[id].params[2] ?? 220;
    return { id, rate: snap(base, this.chordAt(at), voice) / base, gain, at };
  }
  /** When and at which rate a cue sounds; kills climb the chord one tone per kill. */
  place(cue: Cue, now: number): Note {
    const at = nextSlot(this.clock, now, TIMING[cue.id].grid);
    return this.noteAt(cue.id, cue.id === "kill" ? this.kills++ : cue.voice, cue.gain, at);
  }
}
```

```ts
import { admit, type Voice } from "./mixer";
import { RECIPES, TIMING, type CueId } from "./recipes";
import type { Note } from "./score";
import type { SoundSettings } from "./settings";

// Buses: sfx and ui → master; music (drone, drums, arpeggio) → duck → master; master → compressor → speakers.

/** Bus gain scale, so full sliders leave headroom for bursts. */
const HEADROOM = 0.5,
  DUCK_LEVEL = 0.4,
  DUCK_TIME = 0.6;
const BUS_OF = { combat: "sfx", flow: "sfx", ui: "ui", drums: "music" } as const;
type Playing = Voice & { source: AudioBufferSourceNode };

export class AudioEngine {
  readonly ambience: Ambience;
  private master: GainNode;
  private bus: Record<"sfx" | "ui" | "music", GainNode>;
  private duckGain: GainNode;
  private buffers = new Map<CueId, AudioBuffer>();
  private voices: Playing[] = [];
  private last = new Map<CueId, number>();
  constructor(readonly ctx: AudioContext) {
    const compressor = ctx.createDynamicsCompressor();
    compressor.threshold.value = -18;
    compressor.ratio.value = 4;
    compressor.connect(ctx.destination);
    this.master = ctx.createGain();
    this.master.connect(compressor);
    this.bus = { sfx: ctx.createGain(), ui: ctx.createGain(), music: ctx.createGain() };
    this.bus.sfx.connect(this.master);
    this.bus.ui.connect(this.master);
    this.duckGain = ctx.createGain();
    this.bus.music.connect(this.duckGain).connect(this.master);
    this.ambience = new Ambience(ctx, this.bus.music, noiseBuffer(ctx));
  }
  volumes(s: SoundSettings) {
    const now = this.ctx.currentTime;
    this.master.gain.setTargetAtTime(s.muted ? 0 : s.master, now, 0.05);
    for (const b of ["sfx", "ui", "music"] as const) this.bus[b].gain.setTargetAtTime(s[b] * HEADROOM, now, 0.05);
  }
  /** Schedules a note at its time (never in the past); false when the mixer refused it. */
  play(note: Note) {
    const spec = RECIPES[note.id],
      now = this.ctx.currentTime,
      at = Math.max(now, note.at);
    this.voices = this.voices.filter((v) => v.end > now);
    const a = admit(note.id, spec, this.voices, this.last.get(note.id), at);
    if (!a.play) return false;
    if (a.steal) {
      a.steal.source.stop();
      this.voices = this.voices.filter((v) => v !== a.steal);
    }
    const buffer = this.buffer(note.id),
      source = this.ctx.createBufferSource(),
      gain = this.ctx.createGain(),
      // Pitched notes stay in tune; unpitched ones vary a little. Render-side only.
      jitter = TIMING[note.id].pitched ? 0 : (spec.params[1] ?? 0.05);
    source.buffer = buffer;
    source.playbackRate.value = note.rate * (1 + jitter * (Math.random() * 2 - 1));
    gain.gain.value = note.gain;
    source.connect(gain).connect(this.bus[BUS_OF[spec.pool]]);
    source.start(at);
    this.voices.push({ cue: note.id, pool: spec.pool, priority: spec.priority, start: at, end: at + buffer.duration / source.playbackRate.value, source });
    this.last.set(note.id, at);
    if (spec.duck) this.duck(at);
    return true;
  }
  private buffer(id: CueId) {
    let b = this.buffers.get(id);
    if (!b) {
      const samples = buildSamples(RECIPES[id].params, this.ctx.sampleRate);
      b = this.ctx.createBuffer(1, Math.max(1, samples.length), this.ctx.sampleRate);
      b.getChannelData(0).set(samples);
      this.buffers.set(id, b);
    }
    return b;
  }
  private duck(at: number) {
    const g = this.duckGain.gain;
    g.cancelScheduledValues(at);
    g.setValueAtTime(g.value, at);
    g.linearRampToValueAtTime(DUCK_LEVEL, at + 0.05);
    g.setTargetAtTime(1, at + DUCK_TIME, 0.3);
  }
}
/** Two seconds of white noise from xorshift32, the same on every load. */
function noiseBuffer(ctx: AudioContext) {
  const b = ctx.createBuffer(1, ctx.sampleRate * 2, ctx.sampleRate),
    data = b.getChannelData(0);
  let x = 0x9e3779b9;
  for (let i = 0; i < data.length; i++) {
    x ^= x << 13;
    x ^= x >>> 17;
    x ^= x << 5;
    data[i] = (x >>> 0) / 0xffffffff - 0.5;
  }
  return b;
}
```

```ts
import { DRONES } from "./drones";
import { hz, padVoicing } from "./harmony";
import type { DroneId, Mood } from "./mood";
// Plays one drone at a time; a theme change crossfades, a level change glides, a chord change retunes it and the
// tone opens or closes its filter.

const FADE = 3,
  GLIDE = 1.5,
  /** Time constant of the glide to a new chord (settles in about 80 ms). */
  RETUNE = 0.08 / 3,
  /** Time constant of a tone change. */
  TONE = 0.5;
interface Layer {
  id: DroneId;
  gain: GainNode;
  filter: BiquadFilterNode;
  depth: GainNode;
  sources: AudioScheduledSourceNode[];
  /** Two detuned oscillators per voice; voice 0 is the bass. */
  voices: OscillatorNode[][];
  level: number;
  tone: number;
}
export class Ambience {
  private layer: Layer | null = null;
  constructor(
    private ctx: AudioContext,
    private out: AudioNode,
    private noise: AudioBuffer,
  ) {}
  set(mood: Mood) {
    const now = this.ctx.currentTime;
    if (this.layer?.id !== mood.drone) {
      if (this.layer) this.fadeOut(this.layer, now);
      this.layer = this.build(mood.drone, now);
    }
    const layer = this.layer;
    if (Math.abs(layer.level - mood.level) < 0.01) return;
    const g = layer.gain.gain;
    g.cancelScheduledValues(now);
    g.setValueAtTime(g.value, now);
    g.setTargetAtTime(mood.level, now, (layer.level < 0 ? FADE : GLIDE) / 3);
    layer.level = mood.level;
  }
  /** Retunes the drone to a chord (MIDI notes): bass on its root, pad from the top of the chord down, from `at`. */
  chord(notes: readonly number[], at: number) {
    const layer = this.layer;
    if (!layer) return;
    const voicing = padVoicing(notes, layer.voices.length);
    layer.voices.forEach((pair, k) => {
      for (const o of pair) o.frequency.setTargetAtTime(hz(voicing[k]), at, RETUNE);
    });
  }
  /** Opens (> 1) or closes (< 1) the pad's filter relative to the theme's own cutoff. */
  tone(factor: number) {
    const layer = this.layer;
    if (!layer || Math.abs(layer.tone - factor) < 0.02) return;
    const now = this.ctx.currentTime,
      cutoff = DRONES[layer.id].cutoff * factor;
    layer.filter.frequency.setTargetAtTime(cutoff, now, TONE);
    layer.depth.gain.setTargetAtTime(cutoff * 0.35, now, TONE);
    layer.tone = factor;
  }
  stop() {
    if (this.layer) this.fadeOut(this.layer, this.ctx.currentTime);
    this.layer = null;
  }
  private fadeOut(layer: Layer, now: number) {
    const g = layer.gain.gain;
    g.cancelScheduledValues(now);
    g.setValueAtTime(g.value, now);
    g.setTargetAtTime(0, now, FADE / 3);
    for (const s of layer.sources) s.stop(now + FADE * 2);
  }
  private build(id: DroneId, now: number): Layer {
    const spec = DRONES[id],
      ctx = this.ctx,
      gain = ctx.createGain(),
      filter = ctx.createBiquadFilter(),
      lfo = ctx.createOscillator(),
      depth = ctx.createGain();
    gain.gain.setValueAtTime(0, now);
    gain.connect(this.out);
    filter.type = "lowpass";
    filter.frequency.value = spec.cutoff;
    filter.Q.value = 0.7;
    filter.connect(gain);
    lfo.frequency.value = spec.lfo;
    depth.gain.value = spec.cutoff * 0.35;
    lfo.connect(depth).connect(filter.frequency);
    const sources: AudioScheduledSourceNode[] = [lfo],
      voices: OscillatorNode[][] = [],
      each = 0.12 / spec.intervals.length;
    for (const semitones of spec.intervals) {
      const pair: OscillatorNode[] = [];
      for (const cents of [-spec.detune, spec.detune]) {
        const o = ctx.createOscillator(),
          g = ctx.createGain();
        o.type = spec.wave;
        o.frequency.value = spec.root * 2 ** (semitones / 12);
        o.detune.value = cents;
        g.gain.value = each;
        o.connect(g).connect(filter);
        sources.push(o);
        pair.push(o);
      }
      voices.push(pair);
    }
    if (spec.noise > 0) {
      const src = ctx.createBufferSource(),
        band = ctx.createBiquadFilter(),
        g = ctx.createGain();
      src.buffer = this.noise;
      src.loop = true;
      band.type = "bandpass";
      band.frequency.value = spec.cutoff * 2;
      band.Q.value = 0.8;
      g.gain.value = spec.noise;
      src.connect(band).connect(g).connect(gain);
      sources.push(src);
    }
    for (const s of sources) s.start(now);
    return { id, gain, filter, depth, sources, voices, level: -1, tone: 1 };
  }
}
```

```ts
  const value = (id: string) => Number($<HTMLInputElement>(id).value);
  function ready() {
    if (!engine) {
      const e = (engine = new AudioEngine(new AudioContext()));
      e.volumes({ ...DEFAULT_SETTINGS, muted: false });
      new Sequencer(() => e.ctx.currentTime, (from, to) => {
        for (const b of schedule(score.clock, score.music, parts, from, to)) {
          if (b.kind === "chord") e.ambience.chord(b.notes, b.at);
          else if (b.kind === "arp") e.play(score.noteAt("arp", b.voice, 1, b.at));
          else e.play(score.noteAt(DRUM_CUE[b.drum].id, 0, DRUM_CUE[b.drum].gain, b.at));
        }
      }).start();
      theme();
    }
    return engine;
  }
  function theme() {
    const e = ready(),
      drone = $<HTMLSelectElement>("sb-theme").value as DroneId,
      now = e.ctx.currentTime;
    e.ambience.set({ drone, level: 0.6 });
    if (score.theme(drone, now)) e.ambience.chord(score.chordAt(now), now);
  }
  /** The two parts from the sliders, as the game derives them from threat, defense and clear time. */
  function measure(playing: boolean) {
    const defense = value("sb-defense") / 100;
    color = colorOf(value("sb-clear"), color);
    score.extent = extentFor(defense);
    parts = { drums: playing ? layersFor(value("sb-threat") / 100, $<HTMLInputElement>("sb-heavy").checked ? 1 : 0) : [], arp: defense >= 0.75, extent: score.extent, color };
    engine?.ambience.tone(toneFor(color, defense));
  }
  function stop() {
    if (timer) clearInterval(timer);
    timer = null;
    measure(false);
  }
  function play() {
    const e = ready();
    stop();
    const kinds = MIXES[$<HTMLSelectElement>("sb-mix").value],
      shooters = Object.values(DEFAULT_CONTENT.towers).filter((d) => !d.placement && kinds.includes(d.attack.kind)),
      speed = Number(document.querySelector<HTMLInputElement>("[name=sb-speed]:checked")!.value),
      end = e.ctx.currentTime + (16 * 4 * 60) / score.clock.bpm;
    let owed = 0,
      next = 0,
      played = 0,
      total = 0;
    timer = setInterval(() => {
      const now = e.ctx.currentTime,
        events: GameEvent[] = [];
      measure(true);
      for (owed += (value("sb-rate") * speed) / 30; owed >= 1; owed--, next++) {
        const d = shooters[next % shooters.length];
        events.push({ type: "shot", tower: d.id as TowerId, from: at, to: at, color: d.color });
        if (next % 5 === 0) events.push({ type: "kill", at, color: d.color, reward: 5 });
      }
      for (const c of cuesFor(events, DEFAULT_CONTENT, "game")) {
        total++;
        if (e.play(score.place(c, now))) played++;
      }
      $("sb-out").textContent = `${played} of ${total} cues played`;
      if (now >= end) stop();
    }, 1000 / 30);
  }
  $("sb-start").onclick = () => void ready().ctx.resume();
  $("sb-theme").onchange = theme;
  $("sb-play").onclick = play;
  $("sb-stop").onclick = stop;
  $("sb-boss").onclick = () => {
    const e = ready();
    e.play(score.place({ id: "boss", voice: 0, gain: 1 }, e.ctx.currentTime));
  };
  for (const id of ["sb-rate", "sb-threat", "sb-defense", "sb-clear"])
    $<HTMLInputElement>(id).oninput = () => {
      $(`${id}-out`).textContent = String(value(id));
      measure(timer !== null);
    };
  root.querySelectorAll<HTMLButtonElement>("[data-play]").forEach((b) => {
    b.onclick = () => {
      const e = ready();
      e.play(score.place({ id: b.dataset.play as CueId, voice: 0, gain: 1 }, e.ctx.currentTime));
    };
  });
  setInterval(() => {
    if (!engine) return;
    const m = score.music,
      chord = score.chordAt(engine.ctx.currentTime).map((n) => NOTES[n % 12]).join(" ");
    $("sb-info").textContent = `${score.clock.bpm} BPM · ${NOTES[m.root % 12]} ${m.mode} · ${m.groove} · ${color} · chord ${chord}`;
  }, 200);
  $("sb-copy").onclick = () => {
    const lines = [`| Cue | ${CRITERIA.map(([c]) => c).join(" | ")} | Notes |`, `| --- |${CRITERIA.map(() => " --- |").join("")} --- |`];
    root.querySelectorAll<HTMLElement>(".sb-row[data-cue]").forEach((r) => {
      const scores = [...r.querySelectorAll<HTMLSelectElement>("select")].map((s) => s.value || "–");
      lines.push(`| ${r.dataset.cue} | ${scores.join(" | ")} | ${r.querySelector<HTMLInputElement>("input")!.value} |`);
    });
    const out = $<HTMLTextAreaElement>("sb-copied");
    out.value = lines.join("\n");
    navigator.clipboard?.writeText(out.value).catch(() => out.select());
  };
}
function row(id: CueId) {
  const r = RECIPES[id],
    select = ([c]: readonly [string, string]) => `<select aria-label="${c} of ${id}"><option></option>${[1, 2, 3, 4, 5].map((n) => `<option>${n}</option>`).join("")}</select>`;
  return `<div class="sb-row" data-cue="${id}"><button data-play="${id}">${id}</button><small title="cap ${r.cap} · gap ${r.gap}s">${gridLabel(id)}</small>${CRITERIA.map(select).join("")}<input placeholder="notes"></div>`;
}
```
